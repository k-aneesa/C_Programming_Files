#include <stdio.h>

int fibonacci(int n);

int main()
{
    int range,i;
    
    printf("Enter the range  : ");
    scanf("%d", &range);
    
    for(i=0; i<range; i++)
    {
        printf("%d  ", fibonacci(i));
        //printf("\n");
    }

    return 0;
}

int fibonacci(int n)
{
    if(n==0 ||n==1)
    return 1;
    return(fibonacci(n - 1)+ fibonacci(n -2));
}
